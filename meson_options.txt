  description: 'Build the `lust` interpreter: LuaJIT with the selected ' +

)

  value: '',
  description: 'Install directory for the .so modules. Defaults to ' +

  'luajit_xcflags',
  type: 'string',
  value: '',
  description: 'Extra CFLAGS used when building the LuaJIT runtime inside ' +

)

  'static',
  type: 'boolean',
  value: false,
  description: 'Link `lust` (and the LuaJIT build tools) fully statically, ' +
    'the C library included; meant for musl (`./dev.py build ' +
    '--musl-static`). No .so modules are built (a static `lust` cannot ' +
    'load them), libraries with lib_embed = false are left out and the ' +
    'LuaJIT FFI cannot resolve C symbols',
)
